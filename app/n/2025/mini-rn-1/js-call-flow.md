      participant User as JavaScript User Code
      participant NM as NativeModule.js
      participant Global as global.nativeCallSyncHook
      participant JSC as JSCExecutor (C++)
      participant MR as ModuleRegistry (C++)
      participant DIM as DeviceInfoModule (C++)
      participant OS as macOS System APIs

      Note over User, OS: Synchronous Native Method Call Flow

      %% JavaScript Layer
      User->>NM: DeviceInfo.getSystemVersion()
      Note over NM: Method wrapper generated by genMethod()
      NM->>NM: Check if sync method (from syncMethods array)

      alt Sync method validation
          NM->>Global: typeof global.nativeCallSyncHook === 'function'
          Global-->>NM: true
      else No sync hook
          NM->>NM: invariant() - throw error
          NM-->>User: Error: "Sync methods not supported"
      end

      %% Direct sync call
      NM->>Global: global.nativeCallSyncHook(moduleID=0, methodID=1, args=[])
      Note over Global: Static callback function installed by JSCExecutor

      %% Native C++ Layer Entry
      Global->>JSC: JSCExecutor::nativeCallSyncHook(moduleID, methodID, args)
      Note over JSC: Instance method via getCurrentInstance()

      %% Parameter conversion
      JSC->>JSC: Convert JSValue → C++ types
      Note over JSC: moduleIdDouble = JSValueToNumber()<br/>methodIdDouble = JSValueToNumber()<br/>argsJson = jsValueToJSONString()

      %% Validation
      JSC->>JSC: Validate m_moduleRegistry exists
      JSC->>MR: hasModule(moduleIdInt=0)
      MR-->>JSC: true

      JSC->>MR: getModuleName(moduleIdInt=0)
      MR-->>JSC: "DeviceInfo"

      %% Module Registry Dispatch
      JSC->>MR: callSerializableNativeHook(moduleId=0, methodId=1, params="[]")

      MR->>MR: validateIds(moduleId, methodId)
      MR->>MR: Get module instance: modules_[0].get()
      MR->>MR: Get method name: methods[1] = "getSystemVersion"

      %% DeviceInfo Module Handling
      alt moduleName == "DeviceInfo"
          MR->>DIM: dynamic_cast<DeviceInfoModule*>(module)
          DIM-->>MR: Valid DeviceInfoModule instance

          alt methodName == "getSystemVersion"
              MR->>DIM: getSystemVersionImpl()

              %% Platform-specific implementation
              DIM->>OS: [NSProcessInfo processInfo]
              OS-->>DIM: NSProcessInfo instance
              DIM->>OS: operatingSystemVersion
              OS-->>DIM: NSOperatingSystemVersion{major, minor, patch}
              DIM->>DIM: Format version string
              DIM-->>MR: "14.6.1"

          else methodName == "getDeviceId"
              MR->>DIM: getDeviceIdImpl()
              DIM->>OS: sysctlbyname("hw.model", ...)
              OS-->>DIM: "Mac16,7"
              DIM-->>MR: "Mac16,7"
          end

          MR-->>JSC: result string
      else Other modules
          MR-->>JSC: "" (empty string - not implemented)
      end

      %% Result conversion and return
      JSC->>JSC: Convert result string → JSValue
      Note over JSC: stringToJSValue(result)
      JSC-->>Global: JSValueRef result
      Global-->>NM: JavaScript value
      NM-->>User: "14.6.1" (immediate return)

      Note over User, OS: Synchronous - No callbacks, no promises, blocking call
